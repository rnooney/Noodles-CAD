async function processBoundaryImport(overwriteAll = false) { const text = document.getElementById('boxJsonTextInput').value.trim(); if (!text) { alert("Please select a KMZ/KML/JSON file or paste data into the text box."); return; } try { const boxes = parseGeoJsonOrArray(text); if (!boxes || boxes.length === 0) { alert("No valid polygon boundary areas found. Ensure the dataset contains polygons with at least 3 perimeter coordinates."); return; } if (overwriteAll) { if (!confirm(`Are you sure you want to OVERWRITE and DELETE all existing box areas with these ${boxes.length} imported area(s)?`)) { return; } // Wait for the deletion to fully resolve before pushing new records await db.ref('boxAreas').remove(); } // Push each boundary and await database confirmation const pushPromises = boxes.map(box => { const newRef = db.ref('boxAreas').push(); return newRef.set({ name: box.name || 'Unnamed Box', color: '#8e44ad', coordinates: box.coordinates, createdAt: Date.now() }); }); await Promise.all(pushPromises); document.getElementById('boxJsonTextInput').value = ''; const fileInput = document.getElementById('boxJsonFileInput'); if (fileInput) fileInput.value = ''; document.getElementById('jsonImportPanel').style.display = 'none'; alert(`Successfully imported and saved ${boxes.length} response box area(s)!`); } catch (err) { console.error("Import processing error:", err); alert("Database write failed: " + err.message); } }
